cmake_minimum_required(VERSION 3.22.1)
project(squarechess)
file(GLOB SF_SOURCES stockfish/src/*.cpp stockfish/src/nnue/*.cpp stockfish/src/nnue/features/*.cpp stockfish/src/syzygy/*.cpp)
list(FILTER SF_SOURCES EXCLUDE REGEX "/main.cpp$")
add_library(squarefish SHARED bridge.cpp ${SF_SOURCES})
target_include_directories(squarefish PRIVATE stockfish/src)
target_compile_features(squarefish PRIVATE cxx_std_17)
target_compile_definitions(squarefish PRIVATE NNUE_EMBEDDING_OFF NDEBUG IS_64BIT)
if(ANDROID_ABI STREQUAL "arm64-v8a")
    target_compile_definitions(squarefish PRIVATE USE_NEON)
elseif(ANDROID_ABI STREQUAL "x86_64")
    target_compile_definitions(squarefish PRIVATE USE_SSE2)
else()
    message(FATAL_ERROR "Unsupported Android ABI: ${ANDROID_ABI}")
endif()
target_compile_options(squarefish PRIVATE -O3 -fexceptions -frtti)
target_link_options(squarefish PRIVATE "-Wl,-z,max-page-size=16384")
target_link_libraries(squarefish android log)
